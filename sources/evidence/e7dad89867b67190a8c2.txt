    private static void TestE5Rebuild()
    {
        var original = BuildE5([new byte[] { 1, 2, 3 }, new byte[] { 4, 5 }, new byte[] { 6, 7, 8, 9 }], 4);
        var archive = E5Archive.Parse("synthetic.e5", original);
        Equal(3, archive.Entries.Count);
        Sequence(new byte[] { 4, 5 }, archive.ReadDecoded(2));
        var changed = archive.Rebuild(new Dictionary<int, byte[]> { [1] = [9, 9] }, [2], [[8, 8, 8]]);
        var parsed = E5Archive.Parse("synthetic.e5", changed);
        Equal(3, parsed.Entries.Count);
        Sequence(new byte[] { 9, 9 }, parsed.ReadDecoded(1));
        Sequence(new byte[] { 6, 7, 8, 9 }, parsed.ReadDecoded(2));
        Sequence(new byte[] { 8, 8, 8 }, parsed.ReadDecoded(3));
        Equal(4, parsed.DirectoryTrailerLength);
    }

    private static void TestRAtlasRoundTrip()
    {
        using var scope = TempScope.Create();
        var descriptors = new List<FrameDescriptor>();
        var decoded = new List<DecodedFrame>();
        for (var side = 0; side < 2; side++)
        for (var index = 0; index < 20; index++)
        {
            var descriptor = new FrameDescriptor("Pmapobj.e5", "memory", side + 1, index, 1, "r-action",
                side == 0 ? "front" : "back", 48, 64);
            descriptors.Add(descriptor);
            decoded.Add(new DecodedFrame(descriptor, Solid(48, 64, Color.FromArgb(255, side * 100, index * 10, 80)),
                Enumerable.Repeat((byte)(side * 20 + index), 48 * 64).ToArray(), new byte[1024]));
        }
        var asset = new LogicalImageAsset("Pmapobj.e5", 0, "R000", ImageCategory.CharacterR, ResourceKind.RComposite,
            descriptors, Array.Empty<string>(), false, false, string.Empty);
        var service = new ReferenceAtlasService();
        var exported = service.Export(scope.Path, asset, decoded, Color.Magenta);
        True(File.Exists(exported.PngPath) && File.Exists(exported.TransparentPngPath) && File.Exists(exported.PromptPath),
            "R 导出必须包含键色版、透明版和提示词。");
        True(File.Exists(exported.ManifestPath) && File.Exists(exported.GuidePath), "R 导出必须生成布局清单和独立动作对照图。");
        using (var atlas = new Bitmap(exported.PngPath)) Equal(new Size(448, 448), atlas.Size);
        using var normalized = service.Normalize(exported.TransparentPngPath, asset, decoded, new AiAtlasNormalizationOptions());
        try
        {
            Equal(40, normalized.Frames.Count);
            for (var index = 0; index < 40; index++)
            {
                Equal(ReferenceAtlasService.PixelHash(decoded[index].Bitmap),
                    ReferenceAtlasService.PixelHash(normalized.Frames[index].Bitmap));
            }
        }
        finally
        {
            foreach (var frame in decoded) frame.Dispose();
        }
    }

    private static void TestSAtlasRoundTrip()
    {
        using var scope = TempScope.Create();
        var decoded = new List<DecodedFrame>();
        foreach (var action in new[] { (Name: "mov", Count: 11, Size: 48), (Name: "atk", Count: 12, Size: 64), (Name: "spc", Count: 5, Size: 48) })
        for (var index = 0; index < action.Count; index++)
        {
            var descriptor = new FrameDescriptor("Unit_" + action.Name + ".e5", "memory", 337, index, 1,
                action.Name, "none", action.Size, action.Size);
            decoded.Add(new DecodedFrame(descriptor, Solid(action.Size, action.Size,
                Color.FromArgb(255, 50, index * 10, action.Count))));
        }
        var mapping = new SLogicalAssetMapping(337, false, [new(1, 337)], 556);
        var asset = new LogicalImageAsset("SComposite", 337, "S_337", ImageCategory.CharacterS, ResourceKind.SComposite,
            decoded.Select(frame => frame.Descriptor).ToArray(), Array.Empty<string>(), false, false, string.Empty) { SMapping = mapping };
        var service = new ReferenceAtlasService();
        var exported = service.Export(scope.Path, asset, decoded, Color.Magenta);
        using (var atlas = new Bitmap(exported.PngPath)) Equal(new Size(348, 348), atlas.Size);
        using (var transparent = new Bitmap(exported.TransparentPngPath)) Equal(new Size(348, 348), transparent.Size);
        True(File.Exists(exported.ManifestPath) && File.Exists(exported.GuidePath), "S 导出必须生成布局清单和独立动作对照图。");
        var plan = new ReferenceAtlasLayoutPlanner().Plan(asset, decoded);
        Equal("s-semantic-size-grid-v3", plan.LayoutId);
        AssertAtlasFrame(plan, 1, "mov", 0, 25, 0, 48);
        AssertAtlasFrame(plan, 1, "mov", 6, 25, 50, 48);
        AssertAtlasFrame(plan, 1, "spc", 0, 25, 100, 48);
        AssertAtlasFrame(plan, 1, "atk", 0, 43, 152, 64);
        AssertAtlasFrame(plan, 1, "atk", 11, 241, 284, 64);
        var prompt = exported.PromptPath;
        True(File.Exists(prompt) && File.ReadAllText(prompt).Contains("S_337.png"), "导出后未生成对应物理编号的形象图提示词。");
        using var imported = service.Normalize(exported.PngPath, asset, decoded, new AiAtlasNormalizationOptions());
        try
        {
            Equal(28, imported.Frames.Count);
            Equal(1, imported.Frames[0].Descriptor.Turn);
            Equal("mov", imported.Frames[0].Descriptor.Action);
            Equal(1, imported.Frames[^1].Descriptor.Turn);
            Equal("atk", imported.Frames[^1].Descriptor.Action);
            Equal(AtlasBackgroundKind.KeyColor, imported.BackgroundKind);
        }
        finally
        {
            foreach (var frame in decoded) frame.Dispose();
        }
    }

    private static void TestSThreeTurnAtlasLayout()
    {
        using var scope = TempScope.Create();
        var frames = new List<DecodedFrame>();
        for (var turn = 1; turn <= 3; turn++)
        foreach (var action in new[] { (Name: "mov", Count: 11, Size: 48), (Name: "spc", Count: 5, Size: 48), (Name: "atk", Count: 12, Size: 64) })
        for (var index = 0; index < action.Count; index++)
        {
            var descriptor = new FrameDescriptor("Unit_" + action.Name + ".e5", "memory", 240 + turn,
                index, turn, action.Name, "none", action.Size, action.Size);
            frames.Add(new DecodedFrame(descriptor, Solid(action.Size, action.Size,
                Color.FromArgb(255, turn * 60, index * 10 % 255, action.Count))));
        }
        var asset = new LogicalImageAsset("SComposite", 1, "S001", ImageCategory.CharacterS, ResourceKind.SComposite,
            frames.Select(frame => frame.Descriptor).ToArray(), Array.Empty<string>(), true, false, string.Empty)
        {
            SMapping = new SLogicalAssetMapping(1, true, [new(1, 241), new(2, 242), new(3, 243)], 556)
        };
        try
        {
            var service = new ReferenceAtlasService();
            var plan = new ReferenceAtlasLayoutPlanner().Plan(asset, frames);
            Equal(3, plan.TurnCount);
            AssertAtlasFrame(plan, 2, "mov", 0, 14, 150, 48);
            AssertAtlasFrame(plan, 2, "atk", 0, 316, 198, 64);
            AssertAtlasFrame(plan, 3, "spc", 4, 214, 400, 48);
            AssertAtlasFrame(plan, 3, "atk", 11, 514, 528, 64);
            var rejected = false;
            try { service.Export(scope.Path, asset, frames, Color.Magenta); }
            catch (InvalidDataException) { rejected = true; }
            True(rejected, "物理S图库导出必须拒绝把三张不同Unit图片合为一个S文件。");
            rejected = false;
            try { ReferenceAtlasContext.Create(asset, frames, 2); }
            catch (InvalidDataException) { rejected = true; }
            True(rejected, "当前物理S参考图上下文必须拒绝跨物理图号聚合。");
        }
        finally { foreach (var frame in frames) frame.Dispose(); }
    }

    private static void TestActualAiAtlasNormalization()
    {
        var attachment = @"<local-path-redacted>";
        if (!File.Exists(attachment)) throw new SkippedTestException("外部剪贴板素材已不存在；确定性 AI 导入由资源分类实际写回矩阵验证。");
        var profile = new ProjectProfileService().Detect(SampleRoot("加强版6.5未加密版"));
        var catalog = new ImageCatalogService();
        var resource = catalog.BuildCatalog(profile).Single(item => item.Category == ImageCategory.CharacterS);
        var asset = catalog.ReadAssets(profile, resource).Single(item => item.LogicalId == 106);
        var reference = catalog.DecodeFrames(profile, asset);
        try
        {
            using var result = new ReferenceAtlasService().Normalize(attachment, asset, reference,
                new AiAtlasNormalizationOptions { Kernel = AtlasResamplingKernel.Area, BackgroundTolerance = 12 });
            Equal(new Size(1254, 1254), result.SourceSize);
            Equal(348, result.TargetCanvasSize);
            True(Math.Abs(result.ScaleFactor - 3.603448275862069) < .000001, "附件倍率识别错误。");
            Equal(AtlasBackgroundKind.OpaqueColor, result.BackgroundKind);
            True(result.HasWarnings, "不透明黑底恢复必须要求人工确认。");
            Equal(28, result.Frames.Count);
            Equal(11, result.Frames.Count(frame => frame.Descriptor.Action == "mov" && frame.Bitmap.Size == new Size(48, 48)));
            Equal(5, result.Frames.Count(frame => frame.Descriptor.Action == "spc" && frame.Bitmap.Size == new Size(48, 48)));
            Equal(12, result.Frames.Count(frame => frame.Descriptor.Action == "atk" && frame.Bitmap.Size == new Size(64, 64)));
            Equal(new Size(348, 348), result.NormalizedAtlas.Size);
            var attackRows = new[] { 0, 4, 8 }.Select(index => result.FrameDiagnostics.Single(item =>
                item.Descriptor.Action == "atk" && item.Descriptor.FrameIndex == index).SourceBounds.Top).ToArray();
            True(attackRows[0] is >= 500 and <= 620 && attackRows[1] is >= 680 and <= 850 &&
                 attackRows[2] is >= 860 and <= 1030 && attackRows.SequenceEqual(attackRows.Order()),
                "实际附件的三条攻击语义行没有按渐进纵向偏移重新配准：" + string.Join(',', attackRows));
            True(!result.HasBlockingIssues, "实际 AI 图未能形成完整 28 帧：" +
                                            string.Join("；", result.FrameDiagnostics.Where(item => item.Severity == AtlasIssueSeverity.Error)
                                                .Select(item => item.Message)));
        }
        finally { foreach (var frame in reference) frame.Dispose(); }
    }

    private static void TestMcallMutationTransaction()
    {
        using var scope = TempScope.Create();
        var sample = new ProjectProfileService().Detect(SampleRoot("加强版6.5未加密版"));
        var source = sample.ResourcePaths.First(pair => pair.Key.StartsWith("Mcall:", StringComparison.OrdinalIgnoreCase)).Value;

        {
            ["Mcall:00"] = target
        };
        if (sample.TryGetResource("Spalet.e5") is { } palettePath) resources["Spalet.e5"] = palettePath;
        var profile = new ProjectProfile(scope.Path, "mcall-copy", ProjectVersion.Ccz65, false, resources,
            Array.Empty<string>());
        var original = File.ReadAllBytes(target);
        var codec = new StrategyAnimationPackageCodec();
        using var package = codec.Decode(profile, original);
        var planeIndex = package.PlaybackSequence[0];
        var planeSize = package.Width * package.Height;
        var planeOffset = package.RecordBytes.Length + planeIndex * planeSize;
        var localOffset = Array.FindIndex(package.PlaneBytes, planeIndex * planeSize, planeSize, value => value != 0);
        if (localOffset < 0) localOffset = planeIndex * planeSize;
        var originalIndex = package.PlaneBytes[localOffset];
        var targetIndex = originalIndex == 0 ? 1 : 0;
        var palette = new PaletteService().Load(profile);
        using var replacement = new Bitmap(package.PlaneFrames[planeIndex]);
        var pixelOffset = localOffset - planeIndex * planeSize;
        replacement.SetPixel(pixelOffset % package.Width, pixelOffset / package.Width, palette[targetIndex]);
        var candidate = codec.RebuildFramePlanes(profile, package,
            new Dictionary<int, Bitmap> { [planeIndex] = replacement });
        True(candidate.AsSpan(planeOffset, planeSize).SequenceEqual(original.AsSpan(planeOffset, planeSize)) == false,
            "目标动画平面应发生变化。");
        Sequence(original[..planeOffset], candidate[..planeOffset]);
        Sequence(original[(planeOffset + planeSize)..], candidate[(planeOffset + planeSize)..]);

        var incoming = Path.Combine(scope.Path, "replacement.animation.bin");
        File.WriteAllBytes(incoming, candidate);
        var history = new HistoryService(Path.Combine(scope.Path, "backup"));
        var mutations = new ImageMutationService(history);
        var result = mutations.Commit(profile, mutations.PlanWholeFileReplace(profile, target, incoming));
        True(result.Success, string.Join(";", result.Diagnostics));
        Sequence(candidate, File.ReadAllBytes(target));
        var undo = history.Undo(profile, result.HistoryId!);
        True(undo.Success, string.Join(";", undo.Diagnostics));
        Sequence(original, File.ReadAllBytes(target));
        var redo = history.Redo(profile, result.HistoryId!);
        True(redo.Success, string.Join(";", redo.Diagnostics));
        Sequence(candidate, File.ReadAllBytes(target));

        var created = Path.Combine(scope.Path, "Mcall99.e5");
        var createPlan = new MutationPlan(Guid.NewGuid().ToString("N"), MutationKind.Append, "创建 Mcall99",
            [new MutationFile(created, candidate, "创建独立动画包", AllowCreate: true)], Array.Empty<string>(), false);
        var create = mutations.Commit(profile, createPlan);
        True(create.Success && File.Exists(created), string.Join(";", create.Diagnostics));
        True(history.Undo(profile, create.HistoryId!).Success && !File.Exists(created), "撤回创建应恢复为文件不存在。");
        True(history.Redo(profile, create.HistoryId!).Success && File.Exists(created), "前进创建应恢复文件。 ");

        var deletePlan = new MutationPlan(Guid.NewGuid().ToString("N"), MutationKind.Delete, "删除 Mcall99",
            [new MutationFile(created, Array.Empty<byte>(), "删除独立动画包", DeleteTarget: true)],
            Array.Empty<string>(), true);
        var delete = mutations.Commit(profile, deletePlan);

    {

        var back = new byte[48 * 64 * 20];
        for (var frame = 0; frame < 20; frame++)
        {
            Array.Fill(front, (byte)(frame + 1), frame * 48 * 64, 48 * 64);
            Array.Fill(back, (byte)(frame + 41), frame * 48 * 64, 48 * 64);
        }
        File.WriteAllBytes(path, BuildE5([front, back], 0));
        var profile = new ProjectProfile(scope.Path, "strip", ProjectVersion.Ccz65, false,
            new Dictionary<string, string> { ["Pmapobj.e5"] = path }, Array.Empty<string>());
        var resource = new ImageResourceDescriptor("Pmapobj.e5", "R 形象", ImageCategory.CharacterR,
            ResourceKind.RComposite, [path], 2, true, true, false, string.Empty, string.Empty);
        var codec = new RResourceCodec();
        var asset = codec.ReadAssets(profile, resource).Single();
        var frames = codec.DecodeFrames(profile, asset).ToList();
        try
        {
            var changed = frames.Single(frame => frame.Descriptor.ImageNumber == 1 && frame.Descriptor.FrameIndex == 7);
            changed.Bitmap.SetPixel(0, 0, new PaletteService().RequireWritePalette(profile)[200]);
            var index = frames.IndexOf(changed);
            frames[index] = new DecodedFrame(changed.Descriptor, changed.Bitmap, null, changed.PaletteBgra);
            var plan = new RsImportService().PlanImport(profile, asset, frames);
            var candidate = E5Archive.Parse(path, plan.Files.Single().CandidateBytes);
            var rewrittenFront = candidate.ReadDecoded(1);
            var rewrittenBack = candidate.ReadDecoded(2);
            var changedOffset = 7 * 48 * 64;
            Equal((byte)200, rewrittenFront[changedOffset]);
            for (var offset = 0; offset < front.Length; offset++)
                if (offset != changedOffset) Equal(front[offset], rewrittenFront[offset]);
            Sequence(back, rewrittenBack);
        }
        finally
        {
            foreach (var frame in frames.DistinctBy(frame => frame.Bitmap)) frame.Dispose();
        }
    }

    private static void ValidateDialogAtScale(Form form, float scale, bool preserveAutoSize = false)
    {
        if (!preserveAutoSize) form.ClientSize = new Size(1280, 720);
        if (Math.Abs(scale - 1f) > .001f) form.Scale(new SizeF(scale, scale));
        form.PerformLayout();
        using var bitmap = new Bitmap(form.ClientSize.Width, form.ClientSize.Height);
        form.DrawToBitmap(bitmap, new Rectangle(Point.Empty, bitmap.Size));
        foreach (var button in Descendants(form).OfType<Button>().Where(button => button.Visible))
        {
            True(button.Width >= Math.Min(button.MinimumSize.Width, button.PreferredSize.Width) && button.Height > 0,
                $"{form.Text} 按钮尺寸不足：{button.Text}");
            if (button.Parent is FlowLayoutPanel parent && !parent.AutoScroll)
                True(button.Right <= parent.ClientSize.Width + 2 && button.Bottom <= parent.ClientSize.Height + 2,
                    $"{form.Text} 按钮超出自适应按钮栏：{button.Text}");
        }
    }

    private static void SavePixelReplacementSnapshot(Form form, string name)
    {
        var output = Environment.GetEnvironmentVariable("IMAGEMANAGER_PIXEL_TEST_OUTPUT");
        if (string.IsNullOrWhiteSpace(output)) return;
        Directory.CreateDirectory(output);
        using var bitmap = new Bitmap(form.Width, form.Height);
        form.DrawToBitmap(bitmap, new Rectangle(Point.Empty, bitmap.Size));
        var path = Path.Combine(output, name);
        bitmap.Save(path, ImageFormat.Png);
        Console.WriteLine($"PIXEL_REPLACEMENT_LAYOUT {name} {bitmap.Width}x{bitmap.Height}");
        var geometry = new List<string> { $"Form Bounds={form.Bounds}; Client={form.ClientSize}; Preferred={form.PreferredSize}; Visible={form.Visible}" };
        foreach (var control in Descendants(form).Where(control => control is Button or Panel or TableLayoutPanel))
            geometry.Add($"{control.GetType().Name}/{control.Name}/{control.Text}: Bounds={control.Bounds}; Client={control.ClientSize}; Preferred={control.PreferredSize}; Visible={control.Visible}; Parent={control.Parent?.GetType().Name}/{control.Parent?.Bounds}; Grandparent={control.Parent?.Parent?.GetType().Name}/{control.Parent?.Parent?.Bounds}");
        File.WriteAllLines(Path.Combine(output, Path.GetFileNameWithoutExtension(name) + "-几何.txt"), geometry);
    }

    private static void ValidatePixelReplacementVisibleBounds(Form form)
    {
        foreach (var button in Descendants(form).OfType<Button>())
        {
            True(button.Visible, $"换色按钮不可见：{button.Name}/{button.Text}; bounds={button.Bounds}; parent={button.Parent!.Bounds}; form={form.ClientSize}");
            var bounds = form.RectangleToClient(button.RectangleToScreen(button.ClientRectangle));
            True(form.ClientRectangle.Contains(bounds),
                $"换色按钮被窗口裁切：{button.Name}/{button.Text}; bounds={bounds}; client={form.ClientSize}; parent={button.Parent!.Bounds}; grandparent={button.Parent.Parent?.Bounds}");
            for (var ancestor = button.Parent; ancestor != null && ancestor != form; ancestor = ancestor.Parent)
            {
                var childBounds = ancestor.RectangleToClient(button.RectangleToScreen(button.ClientRectangle));
                True(ancestor.ClientRectangle.Contains(childBounds),
                    $"换色按钮被容器裁切：{button.Name}/{button.Text}; bounds={childBounds}; ancestor={ancestor.GetType().Name}/{ancestor.ClientSize}");
            }
        }
    }

    private static IEnumerable<Control> Descendants(Control root)
    {
        foreach (Control child in root.Controls)
        {
            yield return child;
            foreach (var descendant in Descendants(child)) yield return descendant;
        }
    }

    private static byte[] BuildE5(IReadOnlyList<byte[]> payloads, int trailer)
    {
        var dataStart = E5Archive.IndexOffset + payloads.Count * E5Archive.EntrySize + trailer;
        var bytes = new byte[dataStart + payloads.Sum(payload => payload.Length)];
        for (var index = 0; index < E5Archive.IndexOffset; index++) bytes[index] = (byte)(index * 17);
        var offset = dataStart;
        for (var index = 0; index < payloads.Count; index++)
        {
            var entry = E5Archive.IndexOffset + index * E5Archive.EntrySize;
            BinaryPrimitives.WriteUInt32BigEndian(bytes.AsSpan(entry, 4), (uint)payloads[index].Length);
            BinaryPrimitives.WriteUInt32BigEndian(bytes.AsSpan(entry + 4, 4), (uint)payloads[index].Length);
            BinaryPrimitives.WriteUInt32BigEndian(bytes.AsSpan(entry + 8, 4), (uint)offset);
            payloads[index].CopyTo(bytes, offset);
            offset += payloads[index].Length;
        }
        return bytes;
    }

    private static Bitmap Solid(int width, int height, Color color)

    private static void AssertAtlasFrame(ReferenceAtlasLayoutPlan plan, int turn, string action, int index,
        int x, int y, int size)
    {
        var frame = plan.Placements.Single(item => item.Frame.Descriptor.Turn == turn &&
                                                  item.Frame.Descriptor.Action == action &&
                                                  item.Frame.Descriptor.FrameIndex == index);
        Equal(x, frame.Bounds.X);
        Equal(y, frame.Bounds.Y);
        Equal(size, frame.Bounds.Width);
        Equal(size, frame.Bounds.Height);
    }

    private static string TransparentPixelHash(Bitmap bitmap)
    {
        var bytes = new byte[checked(bitmap.Width * bitmap.Height * 4)];
        var offset = 0;
        for (var y = 0; y < bitmap.Height; y++)
        for (var x = 0; x < bitmap.Width; x++)
        {
            var color = bitmap.GetPixel(x, y);
            var key = color.G == 0 && color.B == 255 && color.R is 247 or 255;
            if (color.A == 0 || key)
            {
                bytes[offset++] = 0; bytes[offset++] = 0; bytes[offset++] = 0; bytes[offset++] = 0;
            }
            else
            {
                bytes[offset++] = color.R; bytes[offset++] = color.G; bytes[offset++] = color.B; bytes[offset++] = color.A;
            }
        }
        return Convert.ToHexString(SHA256.HashData(bytes));
    }

    private static void Run(string name, Action test)
    {
        try { test(); _passed++; Console.WriteLine("[PASS] " + name); }
        catch (SkippedTestException ex) { _skipped++; Console.WriteLine("[SKIP] " + name + "：" + ex.Message); }
        catch (Exception ex) { _failed++; Console.WriteLine("[FAIL] " + name + "\n       " + ex); }
    }

    private sealed class SkippedTestException(string message) : Exception(message);

    private static void Equal<T>(T expected, T actual)
    {
        if (!EqualityComparer<T>.Default.Equals(expected, actual)) throw new InvalidOperationException($"期望 {expected}，实际 {actual}。");
    }

    private static void Sequence(byte[] expected, byte[] actual)
    {
        if (!expected.SequenceEqual(actual)) throw new InvalidOperationException($"字节不一致：{Convert.ToHexString(expected)} != {Convert.ToHexString(actual)}");
    }

    private static void True(bool condition, string message)
    {
        if (!condition) throw new InvalidOperationException(message);
    }

        }
