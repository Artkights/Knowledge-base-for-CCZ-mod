    private async void Rerun()
    {
        _normalizationCancellation?.Cancel();
        var cancellation = new CancellationTokenSource();
        _normalizationCancellation = cancellation;
        var revision = ++_normalizationRevision;
        var token = cancellation.Token;
        // Each worker owns its reference bitmaps. Closing the dialog must never
        // race the caller disposing the decoded project frames.
        var sourceFrames = _legacyThreeTurns.Checked ? _legacyReferenceFrames ?? _referenceFrames : _referenceFrames;
        var copies = sourceFrames.Select(frame => new DecodedFrame(frame.Descriptor, new Bitmap(frame.Bitmap),
            frame.PaletteIndexes?.ToArray(), frame.PaletteBgra?.ToArray())).ToArray();
        var context = _context with { Frames = copies };
        try
        {
            UseWaitCursor = true;
            _processing = true;
            _animationTimer.Stop();
            _play.Text = "播放";
            _status.Text = "正在后台自动配准并还原像素…";
            UpdateConfirmState();
            _options.Kernel = (AtlasResamplingKernel)_kernel.SelectedIndex;
            _options.BackgroundTolerance = (int)_tolerance.Value;
            _options.OrderedDither = _dither.Checked;
            var options = _options.Clone();
            var next = await Task.Run(() => _service.Normalize(_pngPath, context, options, token), token);
            if (token.IsCancellationRequested || revision != _normalizationRevision || IsDisposed || Disposing)
            {
                next.Dispose();
                return;
            }
            _result?.Dispose();
            _result = next;
            UpdateSourceEditor(next);
            ReplaceImage(_maskPreview, new Bitmap(next.MaskPreview));
            ReplaceImage(_normalizedPreview, new Bitmap(next.NormalizedAtlas));
            _animationIndex = 0;
            ShowAnimationFrame();
            FillDiagnostics();
            var background = next.BackgroundKind switch
            {
                AtlasBackgroundKind.Transparent => "有效 Alpha",
                AtlasBackgroundKind.KeyColor => "键色",
                _ => "不透明纯色（有损恢复）"
            };
            _summary.Text = $"{Path.GetFileName(_pngPath)}：{next.SourceSize.Width}x{next.SourceSize.Height} → " +
                            $"{next.TargetCanvasSize}x{next.TargetCanvasSize}，倍率 {next.ScaleFactor:0.######}x；" +
                            $"背景 {background} #{next.BackgroundColor.R:X2}{next.BackgroundColor.G:X2}{next.BackgroundColor.B:X2}；" +
                            $"必需帧 {next.Frames.Count}。";
            _reviewed.Checked = false;
            _reviewed.Visible = next.HasWarnings;
            _status.Text = next.HasBlockingIssues ? "请在“裁剪与格位”中修正红色问题。" : next.HasWarnings ? "请检查警告后勾选确认。" : "可以写回。";
            _comparisonStatus.Text = "选择下方帧诊断行查看：原始 → AI → 对齐 → 写入结果。";
            _processing = false;
            _successfulRevision = revision;
            UpdateConfirmState();
        }
        catch (OperationCanceledException) { }
        catch (Exception ex)
        {
            if (IsDisposed || Disposing || revision != _normalizationRevision) return;
            _confirm.Enabled = false;
            MessageBox.Show(this, ex.Message, "AI 参考图处理失败", MessageBoxButtons.OK, MessageBoxIcon.Error);
        }
        finally
        {
            foreach (var frame in copies) frame.Dispose();
            if (ReferenceEquals(_normalizationCancellation, cancellation)) _normalizationCancellation = null;
            cancellation.Dispose();
            if (!IsDisposed && !Disposing && revision == _normalizationRevision)
            {
                UseWaitCursor = false;
                _processing = false;
                UpdateConfirmState();
            }
        }
    }

    private void FillDiagnostics()
    {
        var selectedKey = (_diagnostics.CurrentRow?.Tag as AiAtlasFrameDiagnostic)?.FrameKey;
        _fillingDiagnostics = true;
        try
        {
        _diagnostics.Rows.Clear();
        if (_result == null) return;
        foreach (var item in _result.FrameDiagnostics)
        {
            var index = _diagnostics.Rows.Add(item.Descriptor.Turn, item.Descriptor.Action, item.Descriptor.Direction,
                item.Descriptor.FrameIndex, RectangleText(item.SourceBounds), RectangleText(item.TargetBounds),
                RectangleText(item.ReferenceForegroundBounds), RectangleText(item.AiForegroundBounds),
                RectangleText(item.AlignedForegroundBounds), item.Scale.ToString("0.00x"),
                item.BaselineOffset.ToString("0.0"), item.ColorError.ToString("0.0"),
                item.AiColorError.ToString("0.0"),
                item.IsClipped ? "是" : "否", item.Confidence.ToString("P0"), SeverityText(item.Severity), item.Message);
            var row = _diagnostics.Rows[index];
            row.Tag = item;
            row.DefaultCellStyle.BackColor = item.Severity switch
            {
                AtlasIssueSeverity.Error => Color.MistyRose,
                AtlasIssueSeverity.Warning => Color.LemonChiffon,
                _ => SystemColors.Window
            };
        }
        var selectedRow = _diagnostics.Rows.Cast<DataGridViewRow>().FirstOrDefault(row =>
            row.Tag is AiAtlasFrameDiagnostic item && item.FrameKey == selectedKey)
            ?? _diagnostics.Rows.Cast<DataGridViewRow>().FirstOrDefault();
        if (selectedRow != null)
        {
            _diagnostics.CurrentCell = selectedRow.Cells.Cast<DataGridViewCell>().First(cell => cell.Visible);
            selectedRow.Selected = true;
        }
        }
        finally { _fillingDiagnostics = false; }
        LoadSelectedAdjustment();
        ShowSelectedComparison();
        SelectSourceFrame();
    }

    private void UpdateConfirmState()
        => _confirm.Enabled = !_processing && _successfulRevision == _normalizationRevision &&
            _result is { HasBlockingIssues: false } && (!_result.HasWarnings || _reviewed.Checked);

    private void Confirm()
    {
        if (_result == null || !_confirm.Enabled) return;
        ConfirmedFrames = _result.Frames.Select(frame => new DecodedFrame(frame.Descriptor, new Bitmap(frame.Bitmap),
            frame.PaletteIndexes?.ToArray(), frame.PaletteBgra?.ToArray())).ToArray();
        DialogResult = DialogResult.OK;
        Close();
    }

    private void MaskMouseDown(object? sender, MouseEventArgs e)
    {
        if (_result == null || e.Button != MouseButtons.Left) return;
        var point = ImagePoint(_maskPreview, e.Location, _result.SourceSize);
        if (point == null) return;
        if (_maskTool.SelectedIndex == 0)
        {
            _options.BackgroundSample = null;
            _options.BackgroundSamples.Add(point.Value);
            _undoneBackgroundSamples.Clear();
            Rerun();
            return;
        }
        _painting = true;
        _lastStrokePoint = null;
        AddStroke(point.Value);
    }

    private void MaskMouseMove(object? sender, MouseEventArgs e)
    {
        if (!_painting || _result == null || e.Button != MouseButtons.Left) return;
        var point = ImagePoint(_maskPreview, e.Location, _result.SourceSize);
        if (point == null) return;
        if (_lastStrokePoint is { } last && Distance(last, point.Value) < Math.Max(2, (int)_brushRadius.Value / 2)) return;
        AddStroke(point.Value);
    }

    private void MaskMouseUp(object? sender, MouseEventArgs e)
    {
        if (!_painting) return;
        _painting = false;
        Rerun();
    }

    private void AddStroke(Point point)
    {
        _options.MaskStrokes.Add(new AtlasMaskStroke(_maskTool.SelectedIndex == 1, point, (int)_brushRadius.Value));
        _undoneStrokes.Clear();
        _lastStrokePoint = point;
    }

    private void UndoMask()
    {
        if (_options.MaskStrokes.Count == 0) return;
        var stroke = _options.MaskStrokes[^1];
        _options.MaskStrokes.RemoveAt(_options.MaskStrokes.Count - 1);
        _undoneStrokes.Push(stroke);
        Rerun();
    }

    private void RedoMask()
    {
        if (_undoneStrokes.Count == 0) return;
        _options.MaskStrokes.Add(_undoneStrokes.Pop());
        Rerun();
    }

    private void Nudge(int x, int y)
    {
        var key = CurrentScopeKey();
        if (key == null) return;
        var current = _options.Adjustments.GetValueOrDefault(key, AtlasFrameAdjustment.Default);
        _options.Adjustments[key] = current with { OffsetX = current.OffsetX + x, OffsetY = current.OffsetY + y };
        Rerun();
    }

    private void ApplyScale()
    {
        var key = CurrentScopeKey();
        if (key == null) return;
        var current = _options.Adjustments.GetValueOrDefault(key, AtlasFrameAdjustment.Default);
        _options.Adjustments[key] = current with { ScalePercent = (int)_scale.Value };
        Rerun();
    }

    private void ResetAdjustment()
    {
        var key = CurrentScopeKey();
        if (key == null) return;
        _options.Adjustments.Remove(key);
        _scale.Value = 100;
        Rerun();
    }

    private void LoadSelectedAdjustment()
    {
        var key = CurrentScopeKey();
        var value = key == null ? AtlasFrameAdjustment.Default :
            _options.Adjustments.GetValueOrDefault(key, AtlasFrameAdjustment.Default);
        _scale.Value = Math.Clamp(value.ScalePercent, (int)_scale.Minimum, (int)_scale.Maximum);
    }

    private void ShowSelectedComparison()
    {
        if (_result == null || _diagnostics.CurrentRow?.Tag is not AiAtlasFrameDiagnostic item ||
            !_result.FrameComparisons.TryGetValue(item.FrameKey, out var comparison))
        {
            ReplaceImage(_referencePreview, null);
            ReplaceImage(_rawPreview, null);
            ReplaceImage(_alignedPreview, null);
            ReplaceImage(_finalPreview, null);
            return;
        }

        ReplaceImage(_referencePreview, new Bitmap(comparison.ReferenceFrame));
        ReplaceImage(_rawPreview, new Bitmap(comparison.RawAiFrame));
        ReplaceImage(_alignedPreview, new Bitmap(comparison.AlignedAiFrame));
        ReplaceImage(_finalPreview, _comparisonMode.SelectedIndex switch
        {
            1 => BlendImages(comparison.ReferenceFrame, comparison.RawAiFrame, false),
            2 => BlendImages(comparison.RawAiFrame, comparison.FinalCczFrame, true),
            _ => new Bitmap(comparison.FinalCczFrame)
        });
        var report = _result.QuantizationReports.GetValueOrDefault(item.FrameKey);
        _comparisonStatus.Text = report == null
            ? $"帧 {item.Descriptor.FrameIndex}：未使用索引调色板。"
            : $"帧 {item.Descriptor.FrameIndex}：键色 {report.KeyPixelCount}，非键色 {report.NonKeyPixelCount}，" +
              $"平均 Lab 误差 {report.AverageColorError:0.0}，最大 {report.MaximumColorError:0.0}。";
    }

    private string? CurrentScopeKey()
    {
        if (_scope.SelectedIndex == 0) return "all";
        if (_diagnostics.CurrentRow?.Tag is not AiAtlasFrameDiagnostic item) return null;
        return _scope.SelectedIndex == 1
            ? $"section:{item.Descriptor.Turn}:{(item.Descriptor.Action == "atk" ? "atk" : "small")}"
            : "frame:" + AiAtlasNormalizer.FrameKey(item.Descriptor);
    }

    private void ToggleAnimation()
    {
        if (AnimationFrames().Count < 2) return;
        _animationTimer.Enabled = !_animationTimer.Enabled;
        _play.Text = _animationTimer.Enabled ? "暂停" : "播放";
    }

    private void AdvanceAnimation()
    {
        var frames = AnimationFrames();
        if (frames.Count == 0) return;
        _animationIndex = (_animationIndex + 1) % frames.Count;
        ShowAnimationFrame();
    }

    private void ShowAnimationFrame()
    {
        var frames = AnimationFrames();
        if (frames.Count == 0) { ReplaceImage(_animationPreview, null); return; }
        var frame = frames[Math.Clamp(_animationIndex, 0, frames.Count - 1)];
        ReplaceImage(_animationPreview, new Bitmap(frame.Bitmap));
    }

    protected override void Dispose(bool disposing)
    {
        if (disposing)
        {
            ++_normalizationRevision;
            _normalizationCancellation?.Cancel();
            _animationTimer.Dispose();
            _targetToolTip.Dispose();
            _maskPreview.BackgroundImage?.Dispose();
            _result?.Dispose();
            ReplaceImage(_maskPreview, null);
            ReplaceImage(_referencePreview, null);
            ReplaceImage(_rawPreview, null);
            ReplaceImage(_alignedPreview, null);
            ReplaceImage(_finalPreview, null);
            ReplaceImage(_normalizedPreview, null);
            ReplaceImage(_animationPreview, null);
        }
        base.Dispose(disposing);
    }

        Dock = DockStyle.Fill, SizeMode = PictureBoxSizeMode.Zoom, BackColor = Color.FromArgb(38, 40, 43),

    private static void ReplaceImage(PictureBox box, Image? image)
    {
        var old = box.Image;
        box.Image = image;
        old?.Dispose();
    }

    private static Point? ImagePoint(PictureBox box, Point client, Size image)
    {
        if (box.ClientSize.Width <= 0 || box.ClientSize.Height <= 0) return null;
        var scale = Math.Min(box.ClientSize.Width / (double)image.Width, box.ClientSize.Height / (double)image.Height);
        var width = image.Width * scale; var height = image.Height * scale;
        var left = (box.ClientSize.Width - width) / 2; var top = (box.ClientSize.Height - height) / 2;
        if (client.X < left || client.X >= left + width || client.Y < top || client.Y >= top + height) return null;
        return new Point(Math.Clamp((int)((client.X - left) / scale), 0, image.Width - 1),
            Math.Clamp((int)((client.Y - top) / scale), 0, image.Height - 1));
    }

    private static double Distance(Point left, Point right)
        => Math.Sqrt(Math.Pow(left.X - right.X, 2) + Math.Pow(left.Y - right.Y, 2));

    private static Bitmap BlendImages(Bitmap left, Bitmap right, bool difference)
    {
        var width = Math.Min(left.Width, right.Width);
        var height = Math.Min(left.Height, right.Height);
        var output = new Bitmap(width, height, System.Drawing.Imaging.PixelFormat.Format32bppArgb);
        for (var y = 0; y < height; y++)
        for (var x = 0; x < width; x++)
        {
            var a = left.GetPixel(x, y); var b = right.GetPixel(x, y);
            output.SetPixel(x, y, difference
                ? Color.FromArgb(255, Math.Abs(a.R - b.R), Math.Abs(a.G - b.G), Math.Abs(a.B - b.B))
                : Color.FromArgb(255, (a.R + b.R) / 2, (a.G + b.G) / 2, (a.B + b.B) / 2));
        }
        return output;
    }

    private static string RectangleText(Rectangle rectangle)
        => rectangle.IsEmpty ? "-" : $"({rectangle.X},{rectangle.Y}) {rectangle.Width}x{rectangle.Height}";

    private static string SeverityText(AtlasIssueSeverity severity) => severity switch
        { AtlasIssueSeverity.Error => "阻止", AtlasIssueSeverity.Warning => "警告", _ => "正常" };

    private static Size WorkingSize(int width, int height)
    {
        var area = Screen.PrimaryScreen?.WorkingArea ?? new Rectangle(0, 0, width, height);
        return new Size(Math.Min(width, (int)(area.Width * .9)), Math.Min(height, (int)(area.Height * .9)));
    }
