using System.Security.Cryptography;
using CCZModStudio.Models;

namespace CCZModStudio.Core;

public sealed class RangeEngineVersionProfileService
{
    public const string Canonical65Sha256 = "F9A357585F4D77C273A8BADD3E4EF4062983CC81D8040C0CB74768FAB7C04F5F";
    public const string Canonical65ProfileId = "ccz65-unencrypted-f9a-range-v1";

    public RangeEngineVersionProfile Resolve(CczProject project)
    {
        ArgumentNullException.ThrowIfNull(project);
        var path = project.ResolveGameFile("Ekd5.exe");
        if (!File.Exists(path)) return Unknown("EXE_MISSING");
        var sha = Convert.ToHexString(SHA256.HashData(File.ReadAllBytes(path)));
        return ResolveBySha(sha);
    }

    public RangeEngineVersionProfile ResolveBySha(string sha256)
    {
        if (sha256.Equals(Canonical65Sha256, StringComparison.OrdinalIgnoreCase))
            return Canonical65();
        return Unknown("UNSUPPORTED_SHA");
    }

    public RangeVersionVerificationResult Verify(CczProject project, RangeEngineVersionProfile profile)
    {
        ArgumentNullException.ThrowIfNull(project);
        ArgumentNullException.ThrowIfNull(profile);
        var path = project.ResolveGameFile("Ekd5.exe");
        if (!File.Exists(path)) return new RangeVersionVerificationResult { Status = "EXE_MISSING", IsSafe = false };
        var bytes = File.ReadAllBytes(path);
        var sha = Convert.ToHexString(SHA256.HashData(bytes));
        var errors = new List<string>();
        var warnings = new List<string>();
        if (!sha.Equals(profile.ExeSha256, StringComparison.OrdinalIgnoreCase))
            errors.Add($"Ekd5.exe SHA256 不匹配：期望 {profile.ExeSha256}，实际 {sha}。");

        try
        {
            var snapshot = ExecutableAnalysisSnapshotCache.Shared.GetBase(path);
            if (snapshot.PeImage.ImageBase != profile.ImageBase)
                errors.Add($"ImageBase 不匹配：期望 0x{profile.ImageBase:X8}，实际 0x{snapshot.PeImage.ImageBase:X8}。");
            foreach (var (name, address) in profile.StaticAddresses)
            {
                if (!TryRead(snapshot, address, 1, out _))
                    errors.Add($"静态地址 {name}=0x{address:X8} 无法映射到当前 PE。");
            }
            foreach (var (name, signature) in profile.FunctionSignatures)
            {
                if (!profile.StaticAddresses.TryGetValue(name, out var address))
                {
                    errors.Add($"Function signature {name} has no static address.");
                    continue;
                }
                var expectedSignature = EffectPatchByteService.ParseHex(signature);
                if (!TryRead(snapshot, address, expectedSignature.Length, out var actualSignature) || !actualSignature.SequenceEqual(expectedSignature))
                    errors.Add($"Function signature mismatch for {name} at 0x{address:X8}.");
            }
            foreach (var candidate in profile.HookCandidates)
            {
                if (string.IsNullOrWhiteSpace(candidate.ExpectedOldBytesHex))
                {
                    warnings.Add($"Hook 候选 {candidate.ContractId} 尚未登记旧字节，不能正式写入。");
                    continue;
                }
                var expected = EffectPatchByteService.ParseHex(candidate.ExpectedOldBytesHex);
                if (!TryRead(snapshot, candidate.HookAddress, expected.Length, out var actual) || !actual.SequenceEqual(expected))
                    errors.Add($"Hook 候选 {candidate.ContractId} 的旧字节不匹配。");
            }
        }
        catch (Exception ex)
        {
            errors.Add("PE/静态地址验证失败：" + ex.Message);
        }

        return new RangeVersionVerificationResult
        {
            Status = errors.Count == 0 && warnings.Count == 0 ? "VerifiedIdentity" : errors.Count == 0 ? "IdentityOnly" : "Rejected",
            IsSafe = errors.Count == 0 && warnings.Count == 0,
            EngineSha256 = sha,
            Errors = errors,
            Warnings = warnings
        };
    }

    private static RangeEngineVersionProfile Unknown(string reason)
        => new()
        {
            ProfileId = "unknown-" + reason,
            EngineVersion = "unknown",
            ExeSha256 = string.Empty,
            ImageBase = 0x00400000,
            RangePointerTableAddress = 0x004D1400,
            RangePointerCount = 0,
            ReservedRangeIds = new HashSet<byte> { 0xFF }
        };

    private static bool TryRead(ExecutableAnalysisSnapshot snapshot, uint address, int length, out byte[] bytes)
    {
        try
        {
            bytes = snapshot.ReadVirtualRange(address, length).ToArray();
            return bytes.Length == length;
        }
        catch
        {
            bytes = Array.Empty<byte>();
            return false;
        }
    }
}

public sealed class RangeVersionVerificationResult
{
    public string Status { get; init; } = "Rejected";
    public bool IsSafe { get; init; }
    public string EngineSha256 { get; init; } = string.Empty;
    public IReadOnlyList<string> Errors { get; init; } = Array.Empty<string>();
    public IReadOnlyList<string> Warnings { get; init; } = Array.Empty<string>();
    public bool IdentityMatches => Errors.Count == 0;
}
